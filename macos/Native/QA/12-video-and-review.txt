# C2 (design notes 58, 60, 80; boards C2-V1–V5, C2-R1), C3 (C3-02): the Video page
# with the drop strip under the list and the checkboxes, a fast batch of two with the
# end-of-batch sheet, the Review page (lines, dropped, gaps, language and fps of
# each file) and the review window's title. Captures in
# dark and light.
! cp "$DEMO/sintel_ita_hardsub_demo.mp4" "$DEMO/sintel_eng_hardsub_demo.mp4" "$WORK/"
@launch
wait-ready 600
? ^ready
lang it
theme dark
size 1360x840
params fast
inspector-set langHint it-IT
inspector-tab read
tab review
review-state
? ^review 0$
capture $OUT/R-empty-dark.png
? R-empty-dark.png$
tab process
add $WORK/sintel_ita_hardsub_demo.mp4,$WORK/sintel_eng_hardsub_demo.mp4
capture $OUT/V1-dark.png
? V1-dark.png$
theme light
capture $OUT/V1-light.png
? V1-light.png$
theme dark
focus 0
? ^focus sintel_ita_hardsub_demo\.mp4 · checked 2 · finished 0 · Elabora 2 video on$
drawer 0 open
capture $OUT/V2-dark.png
? V2-dark.png$
drawer 0 close
focus
? ^focus - · checked 2 · finished 0 · Elabora 2 video on$
start
? status ready
wait-step 0 ocr 300
? ^ok
capture $OUT/V3-dark.png
? V3-dark.png$
wait-batch 900
? ^batch done · done 2 failed 0 stopped 0 skipped 0
capture-all $OUT/V5-dark.png
? V5-dark.png$
summary-seen
capture $OUT/V-done-dark.png
? V-done-dark.png$
review-state
? ^review 2 · sintel_ita_hardsub_demo\.mp4 lines [0-9]+ dropped [0-9]+ gaps [0-9]+ lang it-IT fps 6 report yes reviewed no
review-select 0
? ^ok · sintel_ita_hardsub_demo\.mp4$
capture $OUT/R1-dark.png
? R1-dark.png$
theme light
capture $OUT/R1-light.png
? R1-light.png$
theme dark
review 0
? ^Revisione · sintel_ita_hardsub_demo\.mp4$
report-close all
lang en
review 1
? ^Review · sintel_eng_hardsub_demo\.mp4$
report-close all
lang it
# Process Again checks the done rows ("verrà rifatto"): they keep their outcome until the next batch.
process-again
checks
? ^header on · sintel_ita_hardsub_demo\.mp4=redo sintel_eng_hardsub_demo\.mp4=redo$
bottom-bar
? ^checked 2 · finished 2 · Elabora 2 video on$
params default
inspector-tab read
