# Design note 73: after a batch the Preview page shows every video that was
# processed (state done), not only the ready ones: its arrows reach them and
# their frames are searched. Also the Process page after the batch and during
# it (circles, steps with one numbers column, the line over the steps).
! cp "$DEMO/sintel_ita_hardsub_demo.mp4" "$DEMO/sintel_eng_hardsub_demo.mp4" "$WORK/"
@launch
wait-ready 600
? ^ready
lang it
params fast
inspector-set langHint it-IT
inspector close
tab process
add $WORK/sintel_ita_hardsub_demo.mp4,$WORK/sintel_eng_hardsub_demo.mp4
capture $OUT/process-new-list.png
? process-new-list.png$
start
? status ready
wait-step 0 ocr 300
? ^ok
capture $OUT/process-running.png
? process-running.png$
wait-batch 900
? ^batch done · done 2 failed 0 stopped 0 skipped 0
capture $OUT/process-finished-sheet.png
? process-finished-sheet.png$
summary-seen
capture $OUT/process-done.png
? process-done.png$
# C3-09: Preview is a page inside Video; a video never searched searches its
# 20 frames by itself; the title band's arrows go to the video below and above.
preview-open 0
? ^ok · sintel_ita_hardsub_demo\.mp4 · searching$
wait-frames 300
? ^20 frames · 1[5-9] with text · 0 with signs
preview-nav
? ^video 1 di 2 · up off · down sintel_eng_hardsub_demo\.mp4$
capture $OUT/preview-ita.png
? preview-ita.png$
preview-video down
? ^ok · sintel_eng_hardsub_demo\.mp4 · searching$
preview-nav
? ^video 2 di 2 · up sintel_ita_hardsub_demo\.mp4 · down off$
wait-frames 300
? ^20 frames · 1[5-9] with text
preview-video up
? ^ok · sintel_ita_hardsub_demo\.mp4$
preview-close
? ^ok · tab process · back off$
# Design note 72: the report follows the app's language at open and when it changes,
# in place (an edit in a text box survives the switch); 71: its English names.
! test -s "$WORK/sintel_ita_hardsub_demo.review.html" && test ! -e "$WORK/sintel_ita_hardsub_demo.revisione.html"
report-open $WORK/sintel_ita_hardsub_demo.review.html
report-js document.documentElement.lang + ' · ' + document.title
? ^it · Revisione
report-run document.querySelector('textarea').value = 'edited by QA'; return 'ok';
? ok
lang en
report-js document.documentElement.lang + ' · ' + document.title + ' · ' + document.querySelector('textarea').value
? ^en · Review.* · edited by QA$
lang it
report-js document.documentElement.lang + ' · ' + document.title
? ^it · Revisione
report-close all
params default
inspector open
